## Goal
One place that says where things live and how Claude is switched, read by every module. Client files,
the run log and (from P2) the database live in a data folder **outside the repository**, by default
`C:\Users\<you>\BizWaveData` (`~/BizWaveData` on a Mac). A `.env` file in the repo folder is read
automatically, but never overrides a real environment variable and is never read during tests.

**Done when:** with nothing configured, the app resolves every path to `~/BizWaveData/...`; a data
folder inside the repository is refused with a clear message; the run log is written to the data
folder; Claude's on/off switches and model come from the settings; and the whole suite passes with a
temporary data folder and no `.env` loaded.

```python
@dataclass(frozen=True)
class Settings:
    data_dir: Path               # BIZWAVE_DATA_DIR; default <home>/BizWaveData; "~" expanded; made absolute
    deals_dir: Path              # data_dir / "deals"
    samples_dir: Path            # data_dir / "samples"   (sample CIMs and ground truth)
    runlog_path: Path            # BIZWAVE_RUNLOG; default data_dir / "runlog.jsonl"
    database_url: str            # BIZWAVE_DATABASE_URL; default "sqlite:///" + (data_dir / "bizwave.db").as_posix()
    no_llm: bool                 # BIZWAVE_NO_LLM: "1", "true", "yes", "on" (any case) -> True; anything else False
    disabled_uses: frozenset[str]  # BIZWAVE_LLM_DISABLED_USES: comma list, spaces trimmed, blanks dropped
    llm_model: str               # BIZWAVE_LLM_MODEL; default "claude-sonnet-5-5"
    has_api_key: bool            # ANTHROPIC_API_KEY present and non-blank. The key itself is never stored.

def load_settings(env: Mapping[str, str] | None = None, home: Path | None = None) -> Settings
    # Pure: builds Settings from `env` (default os.environ) and `home` (default Path.home()).

def settings(dotenv_path: Path | None = None) -> Settings
    # Loads a .env file once per path per process (unless BIZWAVE_LOAD_DOTENV is "0"/"false"/"no"/
    # "off"), then returns load_settings() built fresh from os.environ on every call (no caching, so
    # tests and long-running processes see changes). dotenv_path defaults to <REPO_ROOT>/.env;
    # tests pass a temporary file.

def ensure_data_dirs(s: Settings | None = None) -> Settings
    # Creates data_dir, deals_dir, samples_dir (idempotent). Raises DataFolderError, before creating
    # anything, if data_dir OR runlog_path is the repository folder or inside it. Paths are compared
    # after following links, case-folded on Windows and macOS. Warns (UserWarning) if data_dir, as
    # written or with links followed, is inside a sync folder (any folder name containing a
    # SYNC_FOLDER_MARKERS entry, case-insensitive), because the database must not live in a synced

class DataFolderError(ValueError): ...

REPO_ROOT: Path   # folder holding pyproject.toml, found by walking up from config.py; None when installed
                  # without the source tree (then the inside-repo check is skipped)

.env file:   <REPO_ROOT>/.env, loaded with python-dotenv, override=False (real environment wins).
```
